# PGM/OSRS construction + Calvin deterministic scheduling + ACORN traversal simulation
import bisect, random, math
def build_pgm(keys, eps=64):
    segs=[]; i=0
    n=len(keys)
    while i<n:
        j=i+1
        lo=-1e18; hi=1e18
        while j<n:
            if j-i>1:
                slope = (keys[j]-keys[i])/(j-i)
                lo = max(lo, slope - eps/(j-i))
                hi = min(hi, slope + eps/(j-i))
                if lo>hi: break
            j+=1
        segs.append((keys[i], keys[j-1], j-i))
        i=j
    return segs

def deterministic_schedule(txs, readsets, writesets):
    # Calvin: sequencer orders, execution deterministic, no locking deadlock risk
    order=sorted(range(len(txs)), key=lambda k: hash(txs[k]))
    conflicts={}
    for i in order:
        for j in order:
            if i!=j and (writesets[i] & readsets[j]):
                conflicts.setdefault(i, set()).add(j)
    # Aria-style batching reorder commit if TW conflict free
    committed=[]
    for i in order:
        if not conflicts.get(i): committed.append(i)
    return committed

def acorn_filtered_search(graph, entry, query, predicate, k=10):
    cand=[entry]; visited=set(); res=[]
    while cand:
        node=cand.pop()
        if node in visited: continue
        visited.add(node)
        if predicate(node):
            res.append(node)
            if len(res)>=k: break
        cand.extend(graph.get(node, [])[:8])
    return res
print(build_pgm(sorted(random.sample(range(10_000),1000)),eps=64)[:3])
