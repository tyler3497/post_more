def build_pgm(keys, eps=64):
    segs=[]; i=0
    while i < len(keys):
        j=i; low=-float('inf'); high=float('inf')
        while j < len(keys):
            if j>i:
                slope = (keys[j]-keys[i])/(j-i) if j!=i else 0
                if not (low <= slope <= high):
                    break
            j+=1
            if j-i>1:
