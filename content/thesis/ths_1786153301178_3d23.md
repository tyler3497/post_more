def build_structure_gaussian_splatting(n=1_000_000, eps=64):
    segs=[]
    sorted_keys = sorted(random.sample(range(n*10), n))

def core_gaussian_splatting_encode_decode(vec, codebook_size=1024, depth=4):
    residual = vec[:]
