```python
# Empirical evaluation harness
import numpy as np, time, hashlib
def benchmark(system, workload, n=10000):
    latencies=[]
    for x in workload[:n]:
        t0=time.time()
        y=system.evaluate(x)
        pi=system.prove(x,y)
        assert system.verify(x,y,pi)
        latencies.append((time.time()-t0)*1000)
    p99=np.percentile(latencies,99)
    mean=np.mean(latencies)
    return mean, p99, np.std(latencies)
```
