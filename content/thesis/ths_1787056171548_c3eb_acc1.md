import bisect, random, math, statistics
def build_pgm(keys, eps=64):
    segs=[]; i=0
    while i < len(keys):
        j=i+1
        while j < len(keys):
            if abs((keys[j]-keys[i]) - (j-i)*(keys[j]-keys[j-1])) > eps*2: break
