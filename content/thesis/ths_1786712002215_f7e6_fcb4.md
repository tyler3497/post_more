def tgn_memory_update(mem, msg, t, delta):
    # mem: node memory N×d, time-encode phi(t)

def hawkes_intensity(mu, alphas, times, t, beta=0.7):
    lam = mu
    for ti, a in zip(times, alphas):
        if ti < t: lam += a*math.exp(-beta*(t-ti))
    return lam

def randomized_smoothing_logit(logits, sigma=0.25, n=1000):
    noises = np.random.randn(n, *logits.shape)*sigma
    smoothed = np.mean([logits+noise for noise in noises], axis=0)
